# Max value
max_value <- max(c(table(df$IR_log_2),table(df$MVB_log_2)))
max_ylim_ceiling <- (ceiling(max_value /10) *10) + 10
MIC_y_scale <- scale_y_continuous(breaks =  seq(0, max_ylim_ceiling, by = 50),limits = c(0,max_ylim_ceiling))

# Get median values
add_clade_median_values <- function(dataframe, variable, clade_variable,max_value) {
  clade_I_median <- dataframe[dataframe[[clade_variable]] == "Clade I", variable] %>% median(na.rm = TRUE)
  clade_II_median <- dataframe[dataframe[[clade_variable]] != "Clade I", variable] %>% median(na.rm = TRUE)
  
  list(clade_I = annotate('point',  x = clade_I_median, y = max_ylim_ceiling,  fill =  clade_colors[["Clade I"]], color = clade_colors[["Clade I"]], shape = 25,  size = 2.5), 
       clade_II = annotate('point',  x = clade_II_median, y = max_ylim_ceiling,  fill =  clade_colors[["Clade II"]],color = clade_colors[["Clade II"]], shape = 25,  size = 2.5))
}

IR_intercepts <-  add_clade_median_values(df, "IR_log_2", "clade_I")
MVB_intercepts <-  add_clade_median_values(df, "MVB_log_2", "clade_I")
